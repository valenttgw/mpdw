DMA

dma <- SMA(data.sma, n = 4)
At <- 2*data.sma - dma
Bt <- 2/(4-1)*(data.sma - dma)
data.dma <- At + Bt
data.ramal2 <- c(NA, data.dma)

f <- c()
for (i in 1:n_test) {
  f[i] <- At[length(At)] + Bt[length(Bt)]*(i)

data.gab2 <- cbind(aktual = c(data1.ts),

  Metode = c("Single Moving Average (SMA)", 
             "Double Moving Average (DMA)", 
             "Single Exponential Smoothing (SES)", 
             "Double Exponential Smoothing (DES)"),
  MAPE_Latih = c(
    ifelse(exists("MAPE_train.sma"), MAPE_train.sma, NA),
    ifelse(exists("MAPE_train.dma"), MAPE_train.dma, NA),
    ifelse(exists("MAPE.ses"), MAPE.ses, NA),
    ifelse(exists("MAPE.des"), MAPE.des, NA)
  ),
  MAPE_Uji = c(
    ifelse(exists("MAPE_test.sma"), MAPE_test.sma, NA),
    ifelse(exists("MAPE_test.dma"), MAPE_test.dma, NA),
    ifelse(exists("MAPE_test.ses"), MAPE_test.ses, NA),
    ifelse(exists("MAPE_test.des"), MAPE_test.des, NA)
  )
)
